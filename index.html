function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var stockSheet = ss.getSheetByName("Stock"); var logSheet = ss.getSheetByName("Log"); // 1. เพิ่มยาใหม่เข้าคลังกลาง/ตู้ if (data.action === 'add_drug') { stockSheet.appendRow([data.drugId, data.drugName, data.lotNo, data.expiryDate, data.cabinetQty, data.centralQty, data.minStock]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } // 2. ย้ายยาจากคลังกลางมาใส่ตู้ if (data.action === 'transfer_drug') { var stockData = stockSheet.getDataRange().getValues(); for (var i = 1; i < stockData.length; i++) { if (String(stockData[i][0]) === String(data.drugId) && String(stockData[i][2]) === String(data.lotNo)) { var row = i + 1; var currentCabinet = Number(stockData[i][4]); var currentCentral = Number(stockData[i][5]); var qty = Number(data.qty); if (currentCentral < qty) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "คลังกลางมีไม่พอ" })).setMimeType(ContentService.MimeType.JSON); } stockSheet.getRange(row, 5).setValue(currentCabinet + qty); stockSheet.getRange(row, 6).setValue(currentCentral - qty); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } } } // 3. เบิกยา (ตัดสต็อกตู้ชั้น 2 และบันทึก Log) if (data.action === 'disburse_drug') { var stockData = stockSheet.getDataRange().getValues(); var foundRow = -1; var currentCabinet = 0; for (var i = 1; i < stockData.length; i++) { if (String(stockData[i][0]) === String(data.drugId) && String(stockData[i][2]) === String(data.lotNo)) { foundRow = i + 1; currentCabinet = Number(stockData[i][4]); break; } } if (foundRow === -1 || currentCabinet < data.qty) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "สินค้าในตู้ชั้น 2 ไม่พอเบิก" })).setMimeType(ContentService.MimeType.JSON); } stockSheet.getRange(foundRow, 5).setValue(currentCabinet - Number(data.qty)); var timestamp = Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss"); logSheet.appendRow([timestamp, data.drugId, data.drugName, data.lotNo, data.expiryDate, data.qty, data.requester, data.department, data.note]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } // 4. แก้ไขข้อมูลยา if (data.action === 'edit_drug') { var stockData = stockSheet.getDataRange().getValues(); for (var i = 1; i < stockData.length; i++) { if (String(stockData[i][0]) === String(data.drugId) && String(stockData[i][2]) === String(data.lotNo)) { var row = i + 1; stockSheet.getRange(row, 2).setValue(data.drugName); stockSheet.getRange(row, 4).setValue(data.expiryDate); stockSheet.getRange(row, 5).setValue(data.cabinetQty); stockSheet.getRange(row, 6).setValue(data.centralQty); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } } } // 5. ลบรายการยา if (data.action === 'delete_drug') { var stockData = stockSheet.getDataRange().getValues(); for (var i = stockData.length - 1; i >= 1; i--) { if (String(stockData[i][0]) === String(data.drugId) && String(stockData[i][2]) === String(data.lotNo)) { stockSheet.deleteRow(i + 1); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } } } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { var action = e.parameter.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'getStock') { var sheet = ss.getSheetByName("Stock"); var data = sheet.getDataRange().getValues(); var products = []; for (var i = 1; i < data.length; i++) { if (data[i][0]) { products.push({ id: String(data[i][0]), name: String(data[i][1]), lotNo: String(data[i][2]), expiryDate: data[i][3] ? Utilities.formatDate(new Date(data[i][3]), "Asia/Bangkok", "yyyy-MM-dd") : "", cabinetQty: Number(data[i][4]), centralQty: Number(data[i][5]), minStock: Number(data[i][6]) }); } } return ContentService.createTextOutput(JSON.stringify(products)).setMimeType(ContentService.MimeType.JSON); } // API ดึงประวัติการเบิก if (action === 'getLogs') { var sheet = ss.getSheetByName("Log"); var data = sheet.getDataRange().getValues(); var logs = []; for (var i = 1; i < data.length; i++) { if (data[i][0]) { logs.push({ timestamp: String(data[i][0]), drugId: String(data[i][1]), drugName: String(data[i][2]), lotNo: String(data[i][3]), expiryDate: String(data[i][4]), qty: Number(data[i][5]), requester: String(data[i][6]), department: String(data[i][7]), note: String(data[i][8]) }); } } return ContentService.createTextOutput(JSON.stringify(logs)).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput("API Running").setMimeType(ContentService.MimeType.TEXT); }
